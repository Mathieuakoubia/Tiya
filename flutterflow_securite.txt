rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    // ── Fonctions utilitaires ────────────────────────────────
    function isAuth() {
      return request.auth != null;
    }
    function isOwner(uid) {
      return isAuth() && request.auth.uid == uid;
    }
    function isMember(squadId) {
      return isAuth() &&
        request.auth.uid in get(/databases/$(database)/documents/squads/$(squadId)).data.users;
    }
    function isTwinOf(uid) {
      return isAuth() &&
        get(/databases/$(database)/documents/users/$(request.auth.uid)).data.twinUid == uid;
    }
    function notTooLong(field, maxLen) {
      return field.size() <= maxLen;
    }
    function validTimestamp(ts) {
      return ts == request.time;
    }

    // ── Profils utilisateurs ─────────────────────────────────
    // CORRECTION CRITIQUE (rapport 2026-05-29) :
    // L'ancienne condition `request.auth.uid != null` annulait la restriction
    // et permettait à tout utilisateur connecté de lire n'importe quel profil
    // (cyclePhase, fcmToken, auraState). Corrigé : lecture restreinte au
    // propriétaire, aux membres du même Squad, et à la Twin liée.
    match /users/{userId} {
      allow read: if isAuth() && (
        request.auth.uid == userId ||
        isMember(resource.data.squadId) ||
        isTwinOf(userId)
      );
      allow create: if isOwner(userId) &&
        request.resource.data.keys().hasAll(['prenom', 'email']) &&
        notTooLong(request.resource.data.prenom, 50) &&
        notTooLong(request.resource.data.email, 100);
      allow update: if isOwner(userId) &&
        // champs autorisés seulement — impossible de se donner un rôle admin
        request.resource.data.diff(resource.data).affectedKeys()
          .hasOnly(['prenom', 'squadId', 'twinUid', 'auraState',
                    'cyclePhase', 'lastActive', 'fcmToken']);
      allow delete: if false;
    }

    // ── Squads ───────────────────────────────────────────────
    match /squads/{squadId} {
      allow read: if isMember(squadId);
      allow create: if isAuth() &&
        request.resource.data.keys().hasAll(['name', 'createdBy', 'users']) &&
        notTooLong(request.resource.data.name, 30) &&
        request.resource.data.users.size() == 1 &&
        request.resource.data.users[0] == request.auth.uid;
      allow update: if isMember(squadId) &&
        request.resource.data.diff(resource.data).affectedKeys()
          .hasOnly(['users', 'niveauBouclier', 'intentionWheel',
                    'lastActivity', 'name']) &&
        request.resource.data.users.size() <= 5;
      allow delete: if false;

      // Sous-collection membres (Squad-Pulse énergie)
      match /members/{memberId} {
        allow read: if isMember(squadId);
        allow write: if isOwner(memberId) &&
          request.resource.data.keys().hasOnly(['displayName', 'energy', 'updatedAt']) &&
          request.resource.data.energy is number &&
          request.resource.data.energy >= 0.0 &&
          request.resource.data.energy <= 1.0;
      }
    }

    // ── Sessions de routines Squad ───────────────────────────
    match /squad_sessions/{sessionId} {
      allow create: if isAuth() &&
        request.resource.data.keys().hasAll(['squadId', 'routineName', 'launchedBy']) &&
        isMember(request.resource.data.squadId) &&
        notTooLong(request.resource.data.routineName, 50) &&
        request.resource.data.launchedBy == request.auth.uid;
      allow read, update: if isAuth() &&
        isMember(resource.data.squadId) &&
        // empêche de modifier routineName ou squadId après création
        request.resource.data.diff(resource.data).affectedKeys()
          .hasOnly(['acceptedBy', 'status', 'completedBy', 'completedAt']);
      allow delete: if false;
    }

    // ── Invitations Squad ────────────────────────────────────
    match /squad_invitations/{invitationId} {
      allow create: if isAuth() &&
        request.auth.uid == request.resource.data.fromUid &&
        notTooLong(request.resource.data.squadName, 30) &&
        notTooLong(request.resource.data.fromName, 50);
      allow read, update: if isAuth() && (
        request.auth.uid == resource.data.fromUid ||
        request.auth.uid == resource.data.toUid
      ) &&
        request.resource.data.diff(resource.data).affectedKeys()
          .hasOnly(['status']);
      allow delete: if false;
    }

    // ── Sessions Twin ────────────────────────────────────────
    match /twin_sessions/{sessionId} {
      allow create: if isAuth() &&
        request.resource.data.launchedBy == request.auth.uid &&
        isTwinOf(request.resource.data.partnerUid);
      allow read, update: if isAuth() && (
        request.auth.uid == resource.data.launchedBy ||
        request.auth.uid == resource.data.partnerUid
      );
      allow delete: if false;
    }

    // ── Audio Capsules ───────────────────────────────────────
    match /audio_capsules/{capsuleId} {
      allow create: if isAuth() &&
        request.auth.uid == request.resource.data.senderUid &&
        // Note : sans Hume, la validation du ton se fait côté client
        // avec analyse d'amplitude. La rule impose juste les champs.
        request.resource.data.keys().hasAll(['senderUid', 'recipientUid',
          'squadId', 'audioUrl', 'createdAt']) &&
        // 15 secondes max validé en amont via durée fichier Storage
        isMember(request.resource.data.squadId);
      allow read: if isAuth() &&
        request.auth.uid == resource.data.recipientUid &&
        // unlock uniquement si delivered = true (mis par Cloud Function)
        resource.data.status == 'delivered';
      allow update: if isAuth() &&
        request.auth.uid == resource.data.recipientUid &&
        request.resource.data.diff(resource.data).affectedKeys().hasOnly(['status']);
      allow delete: if false;
    }

    // ── Intentions (Intention Wheel) ─────────────────────────
    match /squad_intentions/{intentionId} {
      allow read: if isAuth() && isMember(resource.data.squadId);
      allow create, update: if isAuth() &&
        request.auth.uid == request.resource.data.userUid &&
        isMember(request.resource.data.squadId) &&
        notTooLong(request.resource.data.mot, 20);
      allow delete: if false;
    }

    // ── Gratitude Garden (Squad) ─────────────────────────────
    match /squad_gratitude/{galetId} {
      allow read: if isAuth() && isMember(resource.data.squadId);
      allow create: if isAuth() &&
        request.auth.uid == request.resource.data.authorUid &&
        isMember(request.resource.data.squadId) &&
        notTooLong(request.resource.data.mot, 30);
      allow update, delete: if false;
    }

    // ── Capsules Savoring (Twin partagé) ─────────────────────
    match /twin_capsules/{capsuleId} {
      allow read: if isAuth() && (
        request.auth.uid == resource.data.user1Uid ||
        request.auth.uid == resource.data.user2Uid
      );
      allow create: if isAuth() && (
        request.auth.uid == request.resource.data.user1Uid ||
        request.auth.uid == request.resource.data.user2Uid
      ) && isTwinOf(
        request.auth.uid == request.resource.data.user1Uid
          ? request.resource.data.user2Uid
          : request.resource.data.user1Uid
      );
      allow update: if isAuth() && (
        request.auth.uid == resource.data.user1Uid ||
        request.auth.uid == resource.data.user2Uid
      ) &&
        request.resource.data.diff(resource.data).affectedKeys()
          .hasOnly(['reponse1', 'reponse2']);
      allow delete: if false;
    }

    // ── Bloc-notes journalier (Gratitude Flash, Savoring Solo) ─
    match /user_notes/{userId}/entries/{entryId} {
      allow read, write: if isOwner(userId) &&
        notTooLong(request.resource.data.contenu, 500);
    }
  }
}

─────────────────────────────────────────────────────────────────
  ⚠  CHOSES À FAIRE DANS FIREBASE CONSOLE EN PLUS DES RULES
─────────────────────────────────────────────────────────────────

  1. Firebase Storage Rules — pour les fichiers audio (Audio-Capsule) :
     match /audio_capsules/{uid}/{file} {
       allow write: if request.auth.uid == uid
         && request.resource.size < 2 * 1024 * 1024  // 2 MB max
         && request.resource.contentType.matches('audio/.*');
       allow read: if request.auth != null;
     }

  2. Firebase App Check → activer pour Android (Play Integrity) et iOS
     (DeviceCheck) afin de bloquer les appels hors-app.

  3. Firebase Authentication → activer uniquement les providers utilisés
     (Google, Email). Désactiver tous les autres.

  4. Firestore → onglet Usage → activer Budget Alert à 80% du seuil.

  5. Cloud Functions → à créer pour Audio-Capsule :
     - onAudioCapsuleCreated : valide la durée (≤15s), met status='delivered'
     - onRoutineCompleted   : incrémente niveauBouclier via FieldValue.increment(1)
     Ces fonctions évitent que le client puisse se marquer "calme" lui-même.

  PARTIE 2 — OWASP MOBILE TOP 10 (2024) — COUVERTURE SOZIA
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  M1 — Improper Credential Usage
  ────────────────────────────────
  RISQUE  : clés API exposées dans le code ou les assets.
  MESURE  : utiliser google-services.json / GoogleService-Info.plist
            (non committé dans Git). Activer Firebase App Check.
            Ne jamais écrire de clé API en dur dans un custom widget.
  FF      : Settings > Firebase > les clés sont gérées par FF automatiquement.

  M2 — Inadequate Supply Chain Security
  ────────────────────────────────────────
  RISQUE  : package pub.dev compromis ou abandonné.
  MESURE  : vérifier la date de dernière publication de chaque package
            (< 12 mois). Épingler les versions exactes (^X.Y.Z).
            Audit : flutter pub outdated avant chaque build de release.
  NOTE    : google_mlkit_face_detection maintenu par Google → OK.

  M3 — Insecure Authentication / Authorization
  ──────────────────────────────────────────────
  RISQUE  : un utilisateur lit/modifie les données d'un autre.
  MESURE  : toutes les Firestore Rules reposent sur request.auth.uid.
            Les fonctions isMember() et isTwinOf() vérifient l'appartenance
            avant toute lecture/écriture sensible.
            Jamais de champ isAdmin ou role modifiable par le client.

  M4 — Insufficient Input / Output Validation
  ─────────────────────────────────────────────
  RISQUE  : injection de contenu, champs trop longs, types incorrects.
  MESURES (côté client — Custom Widget / Custom Action FF) :

    Champ              Max     Validation
    ──────────────────────────────────────────────────────
    prenom             50 car  trim() + regex [A-Za-zÀ-ÿ '-]
    email              100 car email regex standard
    nom de squad       30 car  trim() + alphanumeric + espaces
    mot d'intention    20 car  trim() + liste blanche de mots
    galet gratitude    30 car  trim() + pas de HTML/scripts
    micro-célébration  15 car  trim()
    message twin       30 car  trim()
    contenu note       500 car trim()
    durée audio        15 sec  vérification côté storage rule (2 MB)

  MESURES (Firestore Rules) : notTooLong() sur tous les champs texte
  (voir Partie 1 ci-dessus).

  OWASP A3 (XSS) : pas applicable au natif Flutter. Mais si un WebView
  est utilisé, activer webview_flutter avec allowedSchemes strict.

  M5 — Insecure Communication
  ─────────────────────────────
  RISQUE  : trafic en clair (HTTP, WS).
  MESURE  : Firebase SDK = HTTPS by default. WebSocket = WSS uniquement.
            Supabase Realtime (recommandé pour Twin/Squad) = WSS.
            Interdire HTTP dans AndroidManifest :
              android:usesCleartextTraffic="false"
            iOS : NSAppTransportSecurity non nécessaire si pas d'HTTP custom.

  M6 — Inadequate Privacy Controls
  ──────────────────────────────────
  RISQUE  : stockage de données sensibles (données de cycle, état émotionnel).
  MESURES :
    - cyclePhase stocké uniquement dans users/{uid} (visible par l'utilisatrice seule).
    - Cycle-Companion : consentement explicite AVANT que le cycle soit partagé avec la Twin.
    - auraState : valeur numérique anonyme, pas de diagnostic médical.
    - Pas de données biométriques brutes stockées (rPPG = calcul local, non sauvegardé).
    - Mention RGPD dans les CGU : base légale = consentement, droit à l'effacement.
    - Endpoint d'effacement à créer (Cloud Function deleteAccount).

  M7 — Insufficient Binary Protections
  ──────────────────────────────────────
  RISQUE  : reverse engineering du binaire.
  MESURE  : activer l'obfuscation Dart dans le build de release :
              flutter build apk --obfuscate --split-debug-info=./debug-info
            Ne PAS stocker de logique métier sensible dans le client.

  M8 — Security Misconfiguration
  ────────────────────────────────
  RISQUE  : Firestore en mode ouvert (allow read, write: if true).
  MESURE  : Les rules de la Partie 1 couvrent ce point.
            Vérifier dans Firebase Console > Firestore > Rules Playground.
            Tester les scénarios : lecture sans auth, écriture d'un autre UID.

  M9 — Insecure Data Storage
  ────────────────────────────
  RISQUE  : token d'accès ou données sensibles dans SharedPreferences en clair.
  MESURE  : Firebase Auth gère les tokens de manière sécurisée (Keychain iOS /
            Keystore Android). Ne jamais écrire de token manuellement.
            Si App State FF stocke un squadId ou twinUid : OK, pas sensible.
            Ne pas stocker auraState ou cyclePhase en local (Firestore seulement).

  M10 — Insufficient Cryptography
  ─────────────────────────────────
  RISQUE  : chiffrement maison, clés codées en dur.
  MESURE  : tout est géré par Firebase (TLS 1.3). Ne pas implémenter de
            chiffrement custom. Les fichiers audio dans Storage sont
            chiffrés au repos par Google Cloud Storage.

  PARTIE 3 — RATE LIMITING (limites de débit)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  Firestore n'a pas de rate limiting natif par utilisateur.
  Ces limites sont enforced par Firestore Rules + Cloud Functions.

  RÈGLE GÉNÉRALE : "strict input validation on all public endpoints"
  = chaque write Firestore passe par une rule qui valide les champs.

  Limites spécifiques par fonctionnalité :
  ──────────────────────────────────────────

  Audio-Capsule
    - Max 1 capsule active par paire (senderUid + recipientUid) par 24h
    - Vérifié en Cloud Function onAudioCapsuleCreated :
        const existing = await db.collection('audio_capsules')
          .where('senderUid', '==', senderUid)
          .where('recipientUid', '==', recipientUid)
          .where('createdAt', '>', yesterday)
          .get();
        if (!existing.empty) throw new Error('Rate limit: 1 capsule / 24h');

  Squad Invitations
    - Max 3 invitations pendantes envoyées par un même utilisateur
    - Vérifié en Firestore Rule (limite approximative) ou Cloud Function

  Squad Sessions
    - Max 1 session active (status='waiting' ou 'active') par squad
    - Rule : lire et compter les sessions actives avant create
      (attention : cette opération coûte 1 read — préférer Cloud Function)

  Intention Wheel
    - 1 intention par membre par jour — enforced par document ID :
      ID = squadId_userUid_YYYY-MM-DD (idempotent par construction)

  Energy update (Squad-Pulse)
    - Throttling côté client dans le Custom Widget :
      Ne pas appeler updateMyEnergy() plus d'une fois toutes les 5 secondes.
      Utiliser un Timer côté Dart (debounce).

  Collecte de routines complétées
    - Cloud Function déclenchée sur création d'un doc squad_sessions avec
      status='completed'. Incrémente niveauBouclier via FieldValue.increment(1).
      Idempotent : inclure un champ processedBy: [] pour éviter double comptage.

  PARTIE 4 — IDEMPOTENCE
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  PRINCIPE : une même opération peut être exécutée plusieurs fois sans
  effet supplémentaire (réseau instable, double-tap, retry automatique).

  Patterns utilisés dans SOZIA :

  1. ID déterministe (le plus simple)
     ─────────────────────────────────
     Pour Intention Wheel : l'ID du document = squadId_userUid_date
       ref = db.collection('squad_intentions').doc('${squadId}_${uid}_2026-05-27')
     → Si l'utilisatrice appuie 2 fois, le set() avec merge:true ne crée
       qu'un seul document.

  2. Champ processedBy[] (pour les Cloud Functions)
     ─────────────────────────────────────────────────
     Pour niveauBouclier : la Cloud Function lit d'abord processedBy
     avant d'incrémenter. Si sessionId déjà présent → skip.
     Utiliser une transaction pour éviter les race conditions.

  3. SetOptions(merge: true) sur les documents utilisateur
     ────────────────────────────────────────────────────────
     Toujours utiliser set({...}, SetOptions(merge: true)) pour les
     mises à jour de profil. Ne jamais utiliser set() sans merge
     (risque d'écraser des champs existants).

  4. Vérification avant envoi pour les invitations
     ─────────────────────────────────────────────────
     Avant sendInvite() : vérifier si une invitation pending existe déjà
     (déjà fait dans squad_service.dart). Appliquer le même pattern pour
     twin_invitations.

  5. Audio-Capsule
     ────────────────
     ID = senderUid_recipientUid_date pour garantir l'unicité journalière.
     Le client vérifie avant d'uploader (1 read) puis la Cloud Function
     re-vérifie (double validation).

  PARTIE 5 — OPTIMISATION DES REQUÊTES FIRESTORE
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  A. État éphémère → PAS dans Firestore
  ────────────────────────────────────────
  Ces données changent à la milliseconde. Les stocker dans Firestore
  provoquerait des milliers de lectures/écritures par session.

  Routines concernées          Alternative recommandée
  ──────────────────────────────────────────────────────────────────
  The High-Five (timing ms)    WebSocket / Supabase Realtime
  Lullaby Haptique (gyroscope) WebSocket / Supabase Realtime
  Twin-Coherence (respiration) WebSocket / Supabase Realtime
  Co-Gaze Alpha (rPPG temps réel) WebSocket / Supabase Realtime
  Breath-Drawing (onde)        WebSocket / Supabase Realtime
  Mirror-Breath (onde)         WebSocket / Supabase Realtime
  Squad-Resonance (tap sync 5) WebSocket / Supabase Realtime
  Thermal-Touch (caresse)      WebSocket / Supabase Realtime
  Vocal-Humming Sync (audio)   WebSocket / Supabase Realtime

  → Seuls les états INITIAL (session créée) et FINAL (session complétée)
    sont écrits dans Firestore.

  B. Compteurs → FieldValue.increment()
  ─────────────────────────────────────
  Ne JAMAIS lire le compteur, l'incrémenter en mémoire, puis l'écrire.
  Utiliser TOUJOURS FieldValue.increment(1).

  Champs concernés :
    squads/{id}.niveauBouclier        ← routines complétées du squad
    users/{id}.routinesCompleted      ← compteur total de l'utilisatrice
    squads/{id}.totalCompletions      ← pour Intention Wheel (mots brillent)

  C. App State FlutterFlow pour les IDs fréquents
  ──────────────────────────────────────────────────
  Stocker en App State (non persistant entre sessions) :
    squadId      ← ne pas requêter à chaque changement d'écran
    twinUid      ← idem
    currentAura  ← valeur numérique locale, sync Firestore toutes les 30s

  D. Pagination et limites sur les collections
  ──────────────────────────────────────────────
  Toujours limiter les queries :
    .limit(20)    pour les listes d'utilisateurs (recherche)
    .limit(5)     pour les membres d'un squad
    .limit(10)    pour l'historique des capsules audio
    .limit(50)    pour le jardin de gratitude Squad

  E. Éviter les lectures en cascade (N+1)
  ─────────────────────────────────────────
  Mauvais pattern (squad_service.dart ligne 46) :
    users.map((uid) => db.collection('users').doc(uid).get())
  → Acceptable pour 5 membres max. À remplacer par un champ dénormalisé
    members: [{uid, prenom, avatar}] dans le document squad quand le
    projet grandit.

  F. Désabonnement des StreamBuilder
  ────────────────────────────────────
  Dans chaque Custom Widget, annuler le StreamSubscription dans dispose().
  FlutterFlow gère cela automatiquement pour les StreamBuilder natifs,
  mais PAS dans les custom widgets. Ne pas oublier :
    @override void dispose() { _subscription?.cancel(); super.dispose(); }

  G. Index Firestore à créer dans la console
  ────────────────────────────────────────────
  Ces queries nécessitent un index composite (Firebase le signale dans
  les logs, mais mieux vaut les créer à l'avance) :

    Collection             Fields                    Order
    squad_invitations      toUid ASC, status ASC     —
    squad_invitations      fromUid ASC, status ASC   —
    squad_sessions         squadId ASC, status ASC   createdAt DESC
    audio_capsules         recipientUid ASC, status  createdAt DESC
    squad_intentions       squadId ASC               date DESC

  PARTIE 6 — CE QUI N'EST PAS POSSIBLE SANS INFRASTRUCTURE EXTERNE
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  Ces fonctionnalités NÉCESSITENT un service externe ou du travail
  ultérieur. Elles sont à remplacer par un placeholder en Phase 1/2.

  1. Analyse vocale (ex-Hume AI)
     ────────────────────────────
     POURQUOI IMPOSSIBLE : Hume AI est supprimé du projet. L'analyse
     d'amplitude locale (détection bruit vs silence) est possible
     avec le package record + noise_meter, mais la détection de ton
     émotionnel nécessite un modèle ML propre à SOZIA.
     SOLUTION TEMPORAIRE : l'enregistrement Audio-Capsule accepte tout
     message ≤ 15 secondes sans filtrage de ton. Un bandeau UI prévient
     l'enregistreuse de rester calme. La validation sera ajoutée quand
     l'outil SOZIA sera prêt.

  2. rPPG (fréquence cardiaque via caméra)
     ────────────────────────────────────────
     POURQUOI IMPOSSIBLE : aucun package Flutter pub.dev fiable et
     maintenu pour le rPPG en 2026. Nécessite un modèle TFLite custom.
     SOLUTION TEMPORAIRE : Co-Gaze Alpha utilise un rythme de pulsation
     fixe (0.1 Hz) au lieu de la moyenne cardiaque réelle. L'utilisatrice
     ne voit pas la différence visuellement.

  3. Synchronisation temps réel Twin/Squad (WebSocket)
     ────────────────────────────────────────────────────
     POURQUOI : Firestore n'est pas adapté aux données à la milliseconde.
     SOLUTION : Supabase Realtime (gratuit jusqu'à 200 connexions simultanées)
     ou Firebase Realtime Database (plus simple à configurer depuis FF).
     À implémenter avant de livrer les routines Twin/Squad.
     Service recommandé : Supabase.
     Custom Action FF à créer : initRealtimeChannel(sessionId)

  4. Génération audio binaural dynamique
     ──────────────────────────────────────
     POURQUOI : just_audio ne génère pas des fréquences à la volée.
     SOLUTION TEMPORAIRE : pré-générer 3 fichiers audio binaural
     (fréquences alpha 10Hz, thêta 6Hz, delta 2Hz) à uploader dans
     Firebase Storage. Audio-Binaurale joue le bon fichier selon
     l'état de l'utilisatrice.

  5. Détection casque (Audio-Binaurale)
     ──────────────────────────────────
     SOLUTION : package audio_session (déjà dans les deps) permet
     de détecter headphonesConnected via AudioSession.instance.
